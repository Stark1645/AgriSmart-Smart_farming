1. System Architecture Overview

AgriSmart is built using a 3-tier architecture:

  • Client Tier: Modern ReactJS 19 with Vite, Framer Motion animations, and CSS Modules.
  • API / Integration Layer: Axios HTTP client with JWT Authorization request interceptors and error response interceptors.
  • Server Tier: Java Spring Boot 3.x REST controllers, Spring Security, and Spring Data JPA.
  • Database Tier: MySQL 8.x relational database.

3.2 Axios Implementation & Interceptors (services/api.js)

import axios from 'axios';

const API_BASE_URL = import.meta.env.VITE_API_BASE_URL || 'http://localhost:8080/api';

export const api = axios.create({
  baseURL: API_BASE_URL,
  timeout: 8000,
  headers: { 'Content-Type': 'application/json' },
});

// Request Interceptor: Inject JWT Token
api.interceptors.request.use((config) => {
  const auth = sessionStorage.getItem('agrismart_auth');
  if (auth) {
    const token = JSON.parse(auth)?.token;
    if (token) config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// Modular Services matching Spring Boot Controllers
export const authAPI = {
  login: async (creds) => (await api.post('/auth/login', creds)).data,
  register: async (data) => (await api.post('/auth/register', data)).data,
};

export const farmAPI = {
  getAllFarms: async () => (await api.get('/farms')).data,
  createFarm: async (data) => (await api.post('/farms', data)).data,
};

export const sensorAPI = {
  getTelemetryByFarmId: async (id) => (await api.get(`/sensor-data/${id}`)).data,
};
